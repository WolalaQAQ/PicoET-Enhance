cmake_minimum_required(VERSION 3.22)
project(ptj_source LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
set(PTJ_JPEG_SOURCE "" CACHE PATH "libjpeg-turbo 3.1.2 source directory")
set(PTJ_JPEG_BUILD "" CACHE PATH "Separate configured libjpeg-turbo build directory")
set(PTJ_JPEG_LIBRARY "" CACHE FILEPATH "Built static JPEG library")
foreach(input PTJ_JPEG_SOURCE PTJ_JPEG_BUILD PTJ_JPEG_LIBRARY)
  if(NOT EXISTS "${${input}}")
    message(FATAL_ERROR "Missing ${input}; use build.py to build libjpeg-turbo separately first")
  endif()
endforeach()
add_library(ptj_source SHARED shim.cpp platform.cpp)
target_compile_definitions(ptj_source PRIVATE PTJ_BUILDING_LIBRARY PTJ_IPC_SHIM)
add_library(ipc_layout OBJECT layout.cpp)
add_library(ipc_core OBJECT core/core.cpp)
set(CORE_DEFINITIONS PTJ_BUILDING_LIBRARY
  ptj_start=core_ptj_start ptj_session_id=core_ptj_session_id ptj_submit=core_ptj_submit
  ptj_submit_capture=core_ptj_submit_capture ptj_get_stats=core_ptj_get_stats
  ptj_stop=core_ptj_stop ptj_destroy=core_ptj_destroy)
target_compile_definitions(ipc_core PRIVATE ${CORE_DEFINITIONS})
add_executable(ptj_ipc_daemon daemon.cpp platform.cpp $<TARGET_OBJECTS:ipc_core>)
target_link_libraries(ptj_ipc_daemon PRIVATE "${PTJ_JPEG_LIBRARY}")
set(PTJ_TARGETS ptj_source ipc_layout ipc_core ptj_ipc_daemon)
if(WIN32)
  target_link_libraries(ptj_ipc_daemon PRIVATE ws2_32 bcrypt)
  target_link_libraries(ptj_source PRIVATE bcrypt)
  add_library(ipc_test_probe SHARED checks/test_probe.cpp platform.cpp)
  target_link_libraries(ipc_test_probe PRIVATE bcrypt)
  add_library(ipc_core_test SHARED core/core.cpp)
  target_compile_definitions(ipc_core_test PRIVATE ${CORE_DEFINITIONS} PTJ_CORE_TEST)
  target_link_libraries(ipc_core_test PRIVATE "${PTJ_JPEG_LIBRARY}" ws2_32)
  add_library(camera_source_test SHARED core/core.cpp)
  target_compile_definitions(camera_source_test PRIVATE PTJ_BUILDING_LIBRARY)
  target_link_libraries(camera_source_test PRIVATE "${PTJ_JPEG_LIBRARY}" ws2_32)
  add_library(camera_bridge_test SHARED camera/camera_bridge.cpp camera/host_test_api.cpp)
  list(APPEND PTJ_TARGETS ipc_test_probe ipc_core_test camera_source_test camera_bridge_test)
elseif(ANDROID)
  # Compile the reviewed vendored ShadowHook source, never archived object files.
  # Keep its constructor objects before the adapter's constructor, as in hook/.
  add_subdirectory(../hook "${CMAKE_CURRENT_BINARY_DIR}/hook" EXCLUDE_FROM_ALL)
  set(SH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../hook/third_party/shadowhook-2.0.1/cpp")
  add_library(picoet_camera_jpeg SHARED $<TARGET_OBJECTS:shadowhook_obj>
    camera/camera_payload.cpp camera/camera_bridge.cpp)
  target_include_directories(picoet_camera_jpeg PRIVATE "${SH_DIR}/include")
  target_compile_options(picoet_camera_jpeg PRIVATE -fno-omit-frame-pointer)
  target_link_libraries(picoet_camera_jpeg PRIVATE dl log m)
  target_link_options(picoet_camera_jpeg PRIVATE -Wl,--exclude-libs,ALL -Wl,--no-undefined
    "-Wl,--version-script,${CMAKE_CURRENT_SOURCE_DIR}/camera/exports.map")
  target_link_libraries(ptj_source PRIVATE dl)
  target_link_options(ptj_source PRIVATE -Wl,--exclude-libs,ALL
    "-Wl,--version-script,${CMAKE_CURRENT_SOURCE_DIR}/shim.exports.map")
  target_link_options(ptj_ipc_daemon PRIVATE -Wl,--exclude-libs,ALL)
  list(APPEND PTJ_TARGETS picoet_camera_jpeg)
else()
  message(FATAL_ERROR "Build the Android source, or use Windows for the host harness")
endif()
foreach(t IN LISTS PTJ_TARGETS)
  target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" core
    "${PTJ_JPEG_SOURCE}/src" "${PTJ_JPEG_BUILD}")
  if(MSVC)
    target_compile_definitions(${t} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${t} PRIVATE /W4 /WX /wd4324)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
  endif()
endforeach()
