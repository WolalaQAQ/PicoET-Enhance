cmake_minimum_required(VERSION 3.22)
project(picoet_postprocessing_tests CXX)
enable_testing()
add_executable(panel_protocol_test panel_protocol_test.cpp)
target_compile_features(panel_protocol_test PRIVATE cxx_std_17)
add_test(NAME panel_protocol COMMAND panel_protocol_test)
add_executable(camera_service_test ../../tongue-source/checks/service_test.cpp)
target_compile_features(camera_service_test PRIVATE cxx_std_17)
target_include_directories(camera_service_test PRIVATE ../../tongue-source ../../tongue-source/core)
if(MSVC)
    target_compile_definitions(camera_service_test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME camera_service COMMAND camera_service_test)
add_executable(postprocessing_test postprocessing_test.cpp
    ../src/expression/postprocessing.cpp ../src/expression/numerical.cpp ../src/expression/mouth.cpp)
target_compile_features(postprocessing_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(postprocessing_test PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(postprocessing_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(PICOET_BASELINE_DIR)
    target_sources(postprocessing_test PRIVATE ${PICOET_BASELINE_DIR}/numerical.cpp ${PICOET_BASELINE_DIR}/mouth.cpp)
    target_include_directories(postprocessing_test PRIVATE ${PICOET_BASELINE_DIR}/..)
    target_compile_definitions(postprocessing_test PRIVATE PICOET_TEST_BASELINE)
endif()
add_test(NAME postprocessing COMMAND postprocessing_test)
add_executable(preset_response_test preset_response_test.cpp
    ../src/expression/postprocessing.cpp ../src/expression/numerical.cpp ../src/expression/mouth.cpp)
target_compile_features(preset_response_test PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(preset_response_test PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(preset_response_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
add_test(NAME preset_response COMMAND preset_response_test)
if(ANDROID)
    target_link_options(postprocessing_test PRIVATE -static-libstdc++)
    add_executable(native_postprocessing_test native_postprocessing_test.cpp
        ../src/expression/postprocessing.cpp ../src/expression/numerical.cpp ../src/expression/mouth.cpp)
    target_compile_features(native_postprocessing_test PRIVATE cxx_std_17)
    target_compile_options(native_postprocessing_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    target_link_options(native_postprocessing_test PRIVATE -static-libstdc++)
    target_link_libraries(native_postprocessing_test PRIVATE dl)
endif()
